# Calliope 0.7 engine stack — runs in its OWN venv (userData/calliope07-venv).
#
# This stack can NEVER share a venv with the 0.6.8 stack
# (requirements.calliope.txt): 0.6.8 needs numpy 1.23 / pandas 1.5 /
# Python <=3.11, while 0.7 needs pandas >=2.1.3 / pyomo >=6.8.2 /
# Python >=3.10.
#
# calliope is exact-pinned: there is no stable 0.7.0 release yet and dev
# releases may change the YAML schema — do not loosen this pin without
# re-running scripts/verify_dual_engine.py.
calliope==0.7.0.dev7

# Floor pins matching calliope 0.7.0.dev7's requirements (pip would resolve
# these anyway; kept explicit so failures surface at install, not at runtime).
# pandas is CAPPED below 3.0: pandas >=2.3 requires xarray >=2024.10, which
# conflicts with calliope dev7's xarray <2024.4 pin (verified at runtime:
# data table loading fails with an ImportError under pandas 3.x).
pandas>=2.1.3,<2.3
numpy>=1.26,<2.3
# pyomo is EXACT-pinned to the version calliope 0.7.0.dev7 was released
# against; newer pyomo (6.9+) crashes on calliope's kernel-based backend.
pyomo==6.8.2
xarray>=2024.1,<2024.4
ruamel.yaml>=0.18
pydantic>=2.11.4
netcdf4>=1.2

# HiGHS solver: calliope 0.7 supports HiGHS via highspy directly (not via
# pyomo's APPSI interface). highspy <2.0 is pinned to stay below the version
# that changes its Python API in ways incompatible with pyomo 6.8.2.
highspy>=1.5,<2.0
